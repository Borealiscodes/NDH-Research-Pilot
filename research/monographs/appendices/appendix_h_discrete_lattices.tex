\begin{definition}[$E_8$ Gosset Lattice Root Decomposition]
Let $\Coordinate8D$ denote an $8$-dimensional integer vector space over $\mathbb{Z}^8$. The $240$ minimal root vectors of the $E_8$ Gosset polytope partition strictly into two disjoint algebraic subfamilies:
\begin{enumerate}
  \item \textbf{$D_8$ Grid Roots ($112$ vertices):} Coordinate vectors $v \in \{-1, 0, 1\}^8$ satisfying $\|v\|_2^2 = \sum_{i=1}^8 v_i^2 = 2$.
  \item \textbf{Half-Integer Spinor Roots ($128$ vertices):} Unscaled coordinate vectors $s \in \{-1, 1\}^8$ with an even number of negative entries, satisfying $\|s\|_2^2 = 8$, which lift to $\pm \frac{1}{2}$ coordinates in $\mathbb{Q}^8$.
\end{enumerate}
Because $2 \neq 8$, the subfamilies are mutually exclusive ($D_8 \cap \mathrm{Spinor} = \emptyset$), preventing state vectors from masquerading across lattice branches (verified in \texttt{E8\_Gosset\_Geometry.v}).
\end{definition}

\begin{lemma}[Exact Spinor Fractional Norm Lifting]
Let $\phi : \mathbb{Z}^8 \to \mathbb{Q}^8$ map each integer spinor component $x_i \in \{-1, 1\}$ to the exact rational half-integer $x_i / 2$. For every valid spinor root $s$, the lifted rational norm-squared equals $2/1$ identically:
\begin{equation}
  \|\phi(s)\|_{\mathbb{Q}^8}^2 = \sum_{i=1}^8 \left(\frac{s_i}{2}\right)^2 = \frac{8}{4} = \frac{2}{1}
\end{equation}
under cross-multiplied integer equivalence (verified in \texttt{E8\_Spinor\_Lifting.v}).
\end{lemma}

\begin{theorem}[Scale-Invariant Base-243 Bijection and $\mathbb{Z}_3$ Parity]
Let $\tau = [t_0, \dots, t_{n-1}]$ be a sequence of balanced trits ($t_i \in \{-1, 0, 1\}$). Define the recursive packing and unpacking operators:
\begin{equation}
  \mathrm{pack}(\tau) = \sum_{i=0}^{n-1} (t_i + 1) \cdot 3^i, \qquad \mathrm{unpack}(v, n) = \left[ \left(\left\lfloor \frac{v}{3^i} \right\rfloor \bmod 3\right) - 1 \right]_{i=0}^{n-1}.
\end{equation}
For all $n \in \mathbb{N}$, $\mathrm{unpack}(\mathrm{pack}(\tau), n) = \tau$ holds in $\mathcal{O}(n)$ linear proof complexity without exponential state-tree explosion \cite{barrett2024smt}. Furthermore, for $15$-trit telemetry words, single-trit bit-flips strictly violate the $\mathbb{Z}_3$ parity invariant $\sum_{i=0}^{14} (t_i + 1) \equiv 0 \pmod 3$ (verified in \texttt{Base243\_Bijection.v} and \texttt{Base243\_Ternary\_Parity.v}).
\end{theorem}
