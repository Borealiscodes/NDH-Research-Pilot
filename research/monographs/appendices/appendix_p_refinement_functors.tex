\begin{definition}[Cache-Aligned Memory Bitplane]
Let $\mathcal{B}_{64} = [b_0, b_1, \dots, b_{63}]$ ($b_i \in \{0, \dots, 255\}$) denote a $64$-byte, cache-aligned physical memory frame (\texttt{repr(C, align(64))}) sweeping a POSIX shared-memory ledger segment. A byte-level deserialization map $\mathcal{U}_{\mathrm{byte}} : \mathcal{B}_{64} \times \{0, \dots, 63\} \to \mathcal{T}^5$ extracts $5$ balanced trits at byte offset $k$ via modular division:
\begin{equation}
  \mathcal{U}_{\mathrm{byte}}(\mathcal{B}_{64}, k) = \left[ \left(\left\lfloor \frac{b_k}{3^j} \right\rfloor \bmod 3\right) - 1 \right]_{j=0}^4.
\end{equation}
\end{definition}

\begin{theorem}[Bit-Exact Extraction Refinement Parity]
For any valid $64$-byte block $\mathcal{B}_{64}$ and any valid offset $k < 64$ containing byte value $b_k$, hardware deserialization $\mathcal{U}_{\mathrm{byte}}(\mathcal{B}_{64}, k)$ matches the inductive algebraic decoder $\mathrm{unpack}(b_k, 5)$ identically:
\begin{equation}
  \mathcal{U}_{\mathrm{byte}}(\mathcal{B}_{64}, k) = \mathrm{unpack}(b_k, 5)
\end{equation}
across the OCaml native extraction boundary (\texttt{libchguard\_enclave.so}, compiled via \texttt{ocamlopt -O3} with \texttt{int64} register mapping in \texttt{RuntimeFusion.v} and \texttt{Extraction.v}).
\end{theorem}

\end{proposition}
