\begin{definition}[Directional Ownership and Semantic Mimicry Detection]
To separate genuine causal trajectory evolution from fluent surface-level mimicry, let $\mathbf{s}_{\mathrm{before}}, \mathbf{s}_{\mathrm{pred}}, \mathbf{s}_{\mathrm{after}} \in \mathbb{R}^N$ denote the pre-action state, predicted intention vector, and observed post-action state \cite{stell2026causa}. The \textbf{Directional Ownership} score $\Omega_{\mathrm{dir}} \in [0.25, 1.0]$ blends directional cosine alignment (dominating small movements) with relative distance accuracy (dominating large movements). If $\Omega_{\mathrm{dir}}$ is undefined or falls below the empirical floor ($\Omega_{\mathrm{dir}} < 0.40$), the step is rejected as \emph{Semantic Mimicry} (\texttt{geometry.py}).
\end{definition}

\begin{proposition}[Permutation Baseline Falsifiability]
Because high-dimensional state geometry inflates raw alignment scores artificially, any observed mean ownership $\bar{\Omega}_{\mathrm{obs}}$ across $M$ episodes must be evaluated against a null distribution $\{\bar{\Omega}_{\pi_b}\}_{b=1}^B$ ($B = 1000$) generated by randomly permuting intention-outcome pairings while preserving marginal state geometry \cite{stell2026causa}:
\begin{equation}
  \Delta_{\mathrm{causal}} = \bar{\Omega}_{\mathrm{obs}} - \mathbb{E}_\pi[\bar{\Omega}_\pi], \qquad p = \frac{1 + \sum_{b=1}^B \mathbb{I}(\bar{\Omega}_{\pi_b} \ge \bar{\Omega}_{\mathrm{obs}})}{B + 1}.
\end{equation}
If $\Delta_{\mathrm{causal}}$ fails to clear the shuffled baseline ($p \ge 0.05$), observed alignment is classified as a geometric artefact rather than causal agency.
\end{proposition}

\begin{definition}[Orthogonal Prohibited Subspace Projection]
Let $P_{\mathrm{forbid}} \in \mathbb{R}^{80 \times 80}$ be the orthogonal projector spanning prohibited dual-use and surveillance axis rotations (\texttt{provenance.py}). For any candidate state vector $\mathbf{x} \in \mathbb{R}^{80}$, the humanitarian boundary requires:
\begin{equation}
  \|P_{\mathrm{forbid}} \mathbf{x}\|_2 \le 44.599.
\end{equation}
Violations trigger immediate memory register clearing (\texttt{gc.collect()}) and un-bypassable process termination via \texttt{os.\_exit(15)} (\texttt{0x0F}).
\end{definition}
